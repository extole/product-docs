import { Extole } from "@extole/react-native-sdk";

const extole = new Extole("share.brand.com", "your-app-name");

function ExtoleScreen({ navigation }) {
  const [extoleView, setExtoleView] =
    useState<JSX.Element | null>(null);

useEffect(() => {
  extole.configure(extoleView, setExtoleView, () => {
    navigation.navigate("Promo");
  });
  }, []);

  return <View style={{ flex: 1 }}>{extoleView}</View>;
}

<Stack.Screen name="Promo" component={ExtoleScreen} />

Getting Started

Choose your implementation track

Extole supports multiple developer paths. Choose Extole App + JavaScript, REST API, or Mobile App based on your architecture and team ownership.

Extole App + JavaScript

Best for teams launching quickly with Extole-managed experiences.

  1. Install the JavaScript SDK

    Add Extole core tags and key zones where users interact with your program.

    Set up JavaScript SDK

  2. Set rewards and eligibility rules

    Define reward types, limits, and participant qualification conditions.

    Configure rewards

  3. Configure your program

    Set program placements, campaign logic, and creative behavior.

    Open program configuration guide

  4. Validate fulfillment and webhook QA

    Send rewardable events and verify fulfillment automation in end-to-end tests.

    Review event delivery options

REST API

Best for teams running secure server-to-server integrations and custom experiences.

  1. Set up authentication

    Configure secure credentials and bearer token access for service calls.

    Configure authentication

  2. Send rewardable events

    Post qualifying events and confirm participant journey updates.

    Send first event

  3. Configure program and rewards

    Define reward models and campaign logic that API events will activate.

    Review rewards configuration

  4. Validate fulfillment automation in QA

    Verify reward state updates and webhook callbacks in downstream systems.

    Set up reward webhooks
